	function get_StationList($id){
		global $db;
		$out = array();
		$result = mysql_query("SELECT s.* FROM tube_weblog_stations AS ws, tube_stations AS s WHERE ws.station_id=s.id AND ws.weblog_id=$id ORDER BY s.name ASC",$db);
		while($row = mysql_fetch_array($result)){
			$out[] = "<a href=\"station.php?id=$row[id]\">$row[name]</a>";
		}
		return implode(", ",$out);
	}

	function get_stations_within($x, $y, $range){
		global $db;
		$range_sq = $range * $range;
		$rows = array();
		$result = mysql_query("SELECT * FROM tube_stations WHERE real_x>=$x-$range AND real_x<=$x+$range AND real_y>=$y-$range AND real_y<=$y+$range", $db);
		while($row = mysql_fetch_array($result)){
			$dif_x = abs($x - $row['real_x']);
			$dif_y = abs($y - $row['real_y']);
			if ($dif_x && $dif_y){
				$dist = ($dif_x * $dif_x) + ($dif_y * $dif_y);
				if ($dist <= $range_sq){
					$rows[] = $row;
				}
			}
		}
		return $rows;
	}

	function filter_text($check_all, $check_url){

		$url_blocks = array();
		$all_blocks = array();

		$result = mysql_query("SELECT * FROM tube_blocks ORDER BY id ASC");
		while ($row = mysql_fetch_array($result)){

			if ($row['all']){
				$all_blocks[] = preg_quote($row['word'], '!');
			}else{
				$url_blocks[] = preg_quote($row['word'], '!');
			}
		}

		if (count($all_blocks)){
			$rx = implode('|', $all_blocks);
			if (preg_match("!$rx!i", $check_all)){ exit; }
		}

		if (count($url_blocks)){
			$rx = implode('|', $url_blocks);
			if (preg_match("!$rx!", $check_url)){ exit; }
		}
	}
